rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // =========================================================================
    // FAHAD DEV — Hardened Zero-Trust Firestore Security Rules (nex-ora-top)
    // =========================================================================

    function isSignedIn() {
      return request.auth != null;
    }

    function adminDoc() {
      return get(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    function isAdminActive() {
      return !('active' in adminDoc().data) || adminDoc().data.active == true;
    }

    function isAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
        isAdminActive() &&
        (!('role' in adminDoc().data) ||
          adminDoc().data.role in ['owner', 'admin', 'superadmin', 'super_admin', 'editor', 'viewer']);
    }

    function isOwner() {
      return isAdmin() &&
        ('role' in adminDoc().data) &&
        adminDoc().data.role in ['owner', 'admin', 'superadmin', 'super_admin'];
    }

    function isEditorOrOwner() {
      return isAdmin() &&
        ('role' in adminDoc().data) &&
        adminDoc().data.role in ['owner', 'admin', 'superadmin', 'super_admin', 'editor'];
    }

    function isValidString(val, maxLen) {
      return val is string && val.size() <= maxLen;
    }

    // -------------------------------------------------------------------------
    // Public Website Settings
    // -------------------------------------------------------------------------
    match /settings/{docId} {
      allow read: if docId in [
        'site', 'public', 'hero', 'social', 'announcement',
        'community', 'ads', 'seo', 'maintenance'
      ] || isAdmin();

      allow write: if isOwner() && docId in [
        'site', 'public', 'hero', 'social', 'announcement',
        'community', 'ads', 'seo', 'maintenance'
      ];

      allow write: if isEditorOrOwner() && docId in [
        'hero', 'social', 'announcement', 'community', 'seo', 'maintenance'
      ];
    }

    // -------------------------------------------------------------------------
    // Navigation Menu & Social Links
    // -------------------------------------------------------------------------
    match /menu/{menuId} {
      allow read: if ((!('visible' in resource.data) || resource.data.visible == true)
                  && (!('enabled' in resource.data) || resource.data.enabled == true))
                  || isAdmin();
      allow write: if isEditorOrOwner();
    }

    match /social/{socialId} {
      allow read: if (!('enabled' in resource.data) || resource.data.enabled == true) || isAdmin();
      allow write: if isEditorOrOwner();
    }

    // -------------------------------------------------------------------------
    // Categories & Nested Versions / Links
    // -------------------------------------------------------------------------
    match /categories/{categoryId} {
      allow read: if ((!('visible' in resource.data) || resource.data.visible == true)
                  && (!('published' in resource.data) || resource.data.published == true))
                  || isAdmin();
      allow write: if isEditorOrOwner();

      match /versions/{versionId} {
        allow read: if ((!('visible' in resource.data) || resource.data.visible == true)
                    && (!('published' in resource.data) || resource.data.published == true))
                    || isAdmin();
        allow write: if isEditorOrOwner();

        match /links/{linkId} {
          allow read: if ((!('active' in resource.data) || resource.data.active == true)
                      && (!('enabled' in resource.data) || resource.data.enabled == true))
                      || isAdmin();
          allow write: if isEditorOrOwner();
        }
      }
    }

    // -------------------------------------------------------------------------
    // Users Directory (User Management: Total Users, Active Users)
    // -------------------------------------------------------------------------
    match /users/{userId} {
      allow read: if (isSignedIn() && request.auth.uid == userId) || isAdmin();
      allow create, update: if (isSignedIn() && request.auth.uid == userId &&
        request.resource.data.keys().hasOnly(['name', 'email', 'active', 'lastActiveAt']))
        || isOwner()
        || (isEditorOrOwner() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['active', 'status', 'updatedAt']));
      allow delete: if isOwner();
    }

    // -------------------------------------------------------------------------
    // Admins Registry (Protected Backend Boundary)
    // -------------------------------------------------------------------------
    match /admins/{uid} {
      allow read: if (isSignedIn() && request.auth.uid == uid) || isAdmin();
      allow create: if isOwner() &&
        request.resource.data.keys().hasOnly(['uid', 'role', 'name', 'username', 'email', 'avatar', 'active', 'updatedAt']) &&
        request.resource.data.role in ['owner', 'editor', 'viewer'] &&
        isValidString(request.resource.data.name, 80) &&
        isValidString(request.resource.data.email, 160);
      allow update: if (isOwner() || (isSignedIn() && request.auth.uid == uid &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name', 'avatar', 'updatedAt']))) &&
        (!('role' in request.resource.data) || request.resource.data.role in ['owner', 'admin', 'superadmin', 'super_admin', 'editor', 'viewer']);
      // Never allow an owner to delete their own admin access
      allow delete: if isOwner() && request.auth.uid != uid;
    }

    // -------------------------------------------------------------------------
    // Activity Logs (Admins read; Editors/Owners append-only)
    // -------------------------------------------------------------------------
    match /logs/{logId} {
      allow read: if isAdmin();
      allow create: if isEditorOrOwner() &&
        request.resource.data.keys().hasOnly(['admin', 'adminUid', 'action', 'before', 'after', 'time']) &&
        request.resource.data.adminUid == request.auth.uid &&
        request.resource.data.time == request.time;
      allow update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // Stats & Analytics
    // -------------------------------------------------------------------------
    match /stats/{day} {
      allow read: if isAdmin();
      allow write: if isOwner();
    }

    match /analytics/{eventId} {
      allow read, update, delete: if isAdmin();
      allow create: if
        request.resource.data.keys().hasOnly([
          'type', 'categoryId', 'versionId', 'linkId', 'platform', 'createdAt'
        ]) &&
        request.resource.data.type in ['download_click', 'view_category', 'view_version', 'share_click'] &&
        request.resource.data.createdAt == request.time;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
